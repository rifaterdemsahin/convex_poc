Mutation Definition
convex/messages.ts 
import { mutation } from "./_generated/server";
import { v } from "convex/values";

export const sendMessage = mutation({
  args: {
    text: v.string(),
    author: v.string(),
  },
  handler: async (ctx, args) => {
    return await ctx.db.insert("messages", {
      text: args.text,
      author: args.author,
    });
  },
});

export const deleteMessage = mutation({
  args: { id: v.id("messages") },
  handler: async (ctx, args) => {
    await ctx.db.delete(args.id);
  },
});
React Hook
components/SendForm.tsx 
import { useMutation } from "convex/react";
import { api } from "../convex/_generated/api";

export function SendForm() {
  const sendMessage = useMutation(
    api.messages.sendMessage
  );

  const handleSend = async (text: string) => {
    // Optimistic update fires immediately;
    // rolls back automatically on failure
    await sendMessage({ text, author: "alice" });
  };

  return <input onKeyDown={(e) => {
    if (e.key === "Enter") handleSend(e.currentTarget.value);
  }} />;
}
ACID Guarantees
Atomic — both insert calls succeed or both fail
Consistent — schema validators run before commit
Isolated — mutations on the same doc run serially
Durable — committed writes survive restarts
